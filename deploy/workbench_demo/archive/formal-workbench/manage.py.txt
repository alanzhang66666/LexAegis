def command(args, **kwargs):
    result = subprocess.run(args, capture_output=True, timeout=kwargs.pop('timeout', 120), **kwargs)
    if result.returncode:
        raise RuntimeError('External deployment command failed: ' + args[0])
    return result.stdout

def compose(*args):
    return ['docker', 'compose', '-f', str(COMPOSE), *args]

def save(name, value):
    (HERE / name).write_text(json.dumps(value, ensure_ascii=False, indent=2), encoding='utf-8')

def build():
    expected = {'lexaegis:public-integration-workbench': APP_BASE, 'lexaegis:cloud-gateway-candidate': 'sha256:a2273e6280dab8792219e7cce34abdbe6f687bfe226b718ca5481db4675d7dd9'}
    for tag, value in expected.items():
        if command(['docker', 'image', 'inspect', tag, '--format', '{{.Id}}']).decode().strip() != value:
            raise ValueError('Base image identity differs from the pinned deployment')
    images = {}
    for label, dockerfile, files in [('app', 'Dockerfile', ('formal_workbench.py', 'formal_gateway.py', 'public_proxy.py', 'poll_events.js')), ('connector', 'Dockerfile.connector', ('edge_connector.py', 'model_route_guard.py'))]:
        with tempfile.TemporaryDirectory() as temp, (HERE / ('build-' + label + '.log')).open('wb') as log:
            directory = Path(temp)
            for name in files:
                shutil.copy2(HERE / name, directory / name)
            text = (HERE / dockerfile).read_text()
            (directory / 'Dockerfile').write_text(text)
            tag = 'lexaegis:formal-local-' + label
            result = subprocess.run(['docker', 'build', '-t', tag, str(directory)], stdout=log, stderr=log)
            if result.returncode:
                raise RuntimeError('External image build failed; see local build log')
        images[label] = command(['docker', 'image', 'inspect', tag, '--format', '{{.Id}}']).decode().strip()
    save('images.json', images)
    print('Formal deployment images built')

def export_access():
    raw = command(compose('exec', '-T', 'formal', 'python', '-c', "from pathlib import Path; print(Path('/data/workbench/production/config/accounts.json').read_text())"))
    accounts = json.loads(raw)
    save('accounts.json', accounts)
    rows = ['<LOCALIZED_OR_PRIVATE_TEXT>', '<LOCALIZED_OR_PRIVATE_TEXT>', '']
    for role, account in accounts.items():
        rows.append(role + '\t' + account['username'] + '\t' + account['password'])
    (HERE / '<LOCALIZED_OR_PRIVATE_TEXT>').write_text('\n'.join(rows), encoding='utf-8')

def start():
    route = HERE.parent / 'previous-url-route.json'
    if route.exists() and json.loads(route.read_text(encoding='utf-8')).get('enabled'):
        command([sys.executable, str(HERE.parent / 'document-workbench/manage.py'), 'start'], timeout=230)
        status()
        return
    command([sys.executable, str(INTEGRATION / 'edge_service.py'), 'start'], timeout=120)
    command(compose('up', '-d', 'edge-model', 'model-guard', 'formal-gateway', 'formal', 'tunnel'), timeout=200)
    export_access()
    sys.path.insert(0, str(HERE.parent))
    from public_access import ensure
    ensure('formal')
    status()

def status(quiet=False):
    sys.path.insert(0, str(HERE.parent))
    from public_access import snapshot
    access = snapshot('formal')
    result = {'url': access['url'], 'public_access': access, 'profile': 'production', 'edge': 'live_local_model', 'model': 'MiniCPM5-2B Q4_K_M', 'source_modified': False, 'formal_cloud_dispatch': 'blocked_pending_original_gates', 'integration_demo_url': json.loads((INTEGRATION / 'public-status.json').read_text())['url']}
    route = HERE.parent / 'previous-url-route.json'
    if route.exists() and json.loads(route.read_text(encoding='utf-8')).get('enabled'):
        result.update(public_profile='integration', public_scope='registered_synthetic_document', public_accounts_file=str(HERE.parent / '<LOCALIZED_OR_PRIVATE_TEXT>'), original_production_workspace='preserved; private backend; public URL now serves the complete demo')
    states = {}
    for label in ('formal', 'formal-gateway', 'edge-model', 'model-guard', 'tunnel'):
        state = json.loads(command(['docker', 'inspect', 'lexaegis-formal-workbench-' + label + '-1']))[0]['State']
        states[label] = {'status': state['Status'], 'health': state.get('Health', {}).get('Status')}
    result['containers'] = states
    result['host_model'] = json.loads(command([sys.executable, str(INTEGRATION / 'edge_service.py'), 'status']))
    save('status.json', result)
    if not quiet:
        print(json.dumps(result))
    return result

def stop():
    route = HERE.parent / 'previous-url-route.json'
    if route.exists() and json.loads(route.read_text(encoding='utf-8')).get('enabled'):
        command([sys.executable, str(HERE.parent / 'document-workbench/manage.py'), 'stop'])
    command(compose('stop', 'tunnel', 'formal', 'formal-gateway', 'model-guard', 'edge-model'))
    model = json.loads(command([sys.executable, str(INTEGRATION / 'edge_service.py'), 'status']))
    observed = subprocess.run(['docker', 'inspect', 'lexaegis-document-demo-workbench-1'], capture_output=True)
    document_active = observed.returncode == 0 and json.loads(observed.stdout)[0]['State']['Running']
    if model['running'] and (not document_active):
        command([sys.executable, str(INTEGRATION / 'edge_service.py'), 'stop'])
    print('Formal workbench stopped; shared local model retained when the document chain is running')

def verify():

    status()
