ROOT = Path(__file__).parent
OLD = ROOT.parent / 'public-workbench'
FORMAL = ROOT.parent / 'formal-workbench'
COMPOSE = ROOT / 'compose.json'
BASE = 'sha256:a2273e6280dab8792219e7cce34abdbe6f687bfe226b718ca5481db4675d7dd9'
APP_TAG = 'lexaegis:registered-document-demo'

def cmd(args, **kwargs):
    r = subprocess.run(args, capture_output=True, timeout=kwargs.pop('timeout', 120), **kwargs)
    if r.returncode:
        raise RuntimeError('Deployment command failed: ' + args[0])
    return r.stdout

def compose(*args):
    return ['docker', 'compose', '-f', str(COMPOSE), *args]

def save(name, value):
    (ROOT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2), encoding='utf-8')

def volume(name):
    existing = subprocess.run(['docker', 'volume', 'inspect', name], capture_output=True)
    if existing.returncode == 0:
        if (json.loads(existing.stdout)[0].get('Labels') or {}).get('lexaegis.deployment') != 'registered-document-20261103':
            raise ValueError('Volume ownership mismatch')
    else:
        cmd(['docker', 'volume', 'create', '--label', 'lexaegis.deployment=registered-document-20261103', name])

def provision(name, files, image=BASE):
    volume(name)
    payload = io.BytesIO()
    with tarfile.open(fileobj=payload, mode='w') as archive:
        for filename, content in files.items():
            member = tarfile.TarInfo(filename)
            member.size = len(content)
            member.mode = 384
            archive.addfile(member, io.BytesIO(content))
    code = "import io,tarfile,sys,os,pathlib; a=tarfile.open(fileobj=io.BytesIO(sys.stdin.buffer.read()),mode='r:'); root=pathlib.Path('/provision'); os.chown(root,10001,10001); os.chmod(root,0o700); [( (root/m.name).write_bytes(a.extractfile(m).read()), os.chmod(root/m.name,0o600), os.chown(root/m.name,10001,10001)) for m in a.getmembers() if m.isfile() and '/' not in m.name and m.name not in {'.','..'}]"
    cmd(['docker', 'run', '--rm', '-i', '--network', 'none', '--read-only', '--user', '0:0', '--cap-drop=ALL', '--cap-add=CHOWN', '--cap-add=DAC_OVERRIDE', '--cap-add=FOWNER', '--security-opt=no-new-privileges', '--mount', 'type=volume,src=' + name + ',dst=/provision', '--entrypoint', 'python', image, '-c', code], input=payload.getvalue())

def signed_status():
    code = "from pathlib import Path; import urllib.request; from lexaegis.util import sign; s=Path('/run/private/gateway-secret').read_text().strip(); r=urllib.request.Request('http://127.0.0.1:8766/deployment/status',headers={'X-Gateway-Signature':sign({'op':'health'},s)}); print(urllib.request.urlopen(r,timeout=5).read().decode())"
    return json.loads(cmd(['docker', 'exec', 'lexaegis-cloud-integration-gateway-1', 'python', '-c', code]))

def build():
    candidate = ROOT / 'candidate'
    if cmd(['docker', 'image', 'inspect', 'lexaegis:cloud-gateway-candidate', '--format', '{{.Id}}']).decode().strip() != BASE:
        raise ValueError('Pinned local base image changed')
    dockerfile = 'FROM lexaegis:cloud-gateway-candidate\nUSER 0:0\nCOPY candidate/lexaegis /app/lexaegis\nCOPY candidate/packs /app/packs\nCOPY candidate/frontend/dist /app/frontend/dist\nCOPY deployment /opt/deployment\nRUN mkdir -p /data/workbench && chown -R 10001:10001 /data\nENV PYTHONPATH=/opt/deployment:/opt/cloud_gateway:/app PYTHONDONTWRITEBYTECODE=1 PYTHONUNBUFFERED=1\nUSER 10001:10001\nCMD ["python","/opt/deployment/document_workbench.py"]\n'
    with tempfile.TemporaryDirectory() as temp, (ROOT / 'build.log').open('wb') as log:
        context = Path(temp)
        for name in ('lexaegis', 'packs', 'frontend/dist'):
            shutil.copytree(candidate / name, context / 'candidate' / name, ignore=shutil.ignore_patterns('__pycache__', '*.pyc'))
        deployment = context / 'deployment'
        deployment.mkdir()
        for name in ('document_workbench.py', 'ds_gateway.py', 'shared_budget.py', 'registered-manifest.json'):
            shutil.copy2(ROOT / name, deployment / name)
        for name in ('public_proxy.py', 'poll_events.js'):
            shutil.copy2(OLD / name, deployment / name)
        (context / 'Dockerfile').write_text(dockerfile)
        result = subprocess.run(['docker', 'build', '-t', APP_TAG, str(context)], stdout=log, stderr=log)
        if result.returncode:
            raise RuntimeError('Build failed; inspect private build.log')
    image = cmd(['docker', 'image', 'inspect', APP_TAG, '--format', '{{.Id}}']).decode().strip()
    save('image.json', {'image_config_id': image, 'original_repository_modified': False})
    print('Registered document image built')

def configure():

        old = signed_status()

        if old['quota']['limit'] != 20 or used >= 20:
            raise ValueError('No remaining user-authorized trial budget')
        save('global-policy.json', {'initial_reserved': used, 'lifetime_limit': 20, 'authorization': 'Existing user grant: previous ten plus at most ten additional; shared by all synthetic gateways', 'created_at': datetime.now(timezone.utc).isoformat()})

    secret = ROOT / 'gateway-secret'
    if not secret.exists():
        secret.write_text(secrets.token_urlsafe(48))

    provision(prefix + 'signing', {'gateway-secret': signing})

def start():
    cmd(['docker', 'compose', '-f', str(OLD / 'cloud.compose.json'), 'up', '-d', '--no-deps', 'gateway'], timeout=120)
    for _ in range(45):
        try:
            if signed_status()['quota']['limit'] == 20:
                break
        except RuntimeError:
            pass
        time.sleep(0.5)
    else:
        raise RuntimeError('Existing gateway did not become ready')
    cmd(compose('up', '-d', 'edge-model', 'model-guard', 'doc-gateway', 'workbench', 'tunnel'), timeout=220)
    for _ in range(90):
        try:
            data = json.loads(cmd(['docker', 'exec', 'lexaegis-document-demo-workbench-1', 'python', '-c', "from pathlib import Path; print(Path('/data/workbench/integration/config/accounts.json').read_text())"]))
            save('accounts.json', data)
            labels = {'admin': '<LOCALIZED_OR_PRIVATE_TEXT>', 'case_operator': '<LOCALIZED_OR_PRIVATE_TEXT>', 'technical_reviewer': '<LOCALIZED_OR_PRIVATE_TEXT>', 'legal_reviewer': '<LOCALIZED_OR_PRIVATE_TEXT>', 'approver': '<LOCALIZED_OR_PRIVATE_TEXT>'}
            (ROOT / '<LOCALIZED_OR_PRIVATE_TEXT>').write_text('<LOCALIZED_OR_PRIVATE_TEXT>' + '\n'.join((labels[k] + '\t' + v['username'] + '\t' + v['password'] for k, v in data.items())), encoding='utf-8')
            break
        except (RuntimeError, ValueError):
            time.sleep(0.5)
    status()

def status():
    logs = cmd(compose('logs', '--no-color', '--tail', '160', 'tunnel')).decode(errors='replace')
    urls = re.findall('https://[a-z0-9-]+\\.trycloudflare\\.com', logs)
    states = {}
    for label in ('workbench', 'doc-gateway', 'edge-model', 'model-guard', 'tunnel'):
        info = json.loads(cmd(['docker', 'inspect', 'lexaegis-document-demo-' + label + '-1']))[0]
        states[label] = {'status': info['State']['Status'], 'health': info['State'].get('Health', {}).get('Status'), 'image': info['Image']}
    result = {'url': urls[-1] if urls else None, 'profile': 'integration', 'scope': 'registered_synthetic_document', 'quota': signed_status()['quota'], 'original_repository_modified': False, 'production_dispatch': 'blocked', 'containers': states}
    save('status.json', result)
    print(json.dumps(result))

def stop():
    cmd(compose('stop', 'tunnel', 'workbench', 'doc-gateway', 'model-guard', 'edge-model'))
    print('Document demo stopped; existing workspaces, host model and all data retained')
if __name__ == '__main__':
    p = argparse.ArgumentParser()
    p.add_argument('action', choices=['build', 'configure', 'start', 'status', 'stop'])
    a = p.parse_args()
    try:
        {'build': build, 'configure': configure, 'start': start, 'status': status, 'stop': stop}[a.action]()
    except Exception as e:
        print('Deployment stopped: ' + str(e))
        raise SystemExit(1)
