"""Assemble a report from receipts and read-only verification, without dispatch."""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import shutil
import xml.etree.ElementTree as ET
ROOT = Path(__file__).parent

def read(name):
    return json.loads((ROOT / name).read_text(encoding='utf-8'))

def tests(name):
    suite = ET.parse(ROOT / name).getroot().find('testsuite')
    result = {key: int(suite.get(key, '0')) for key in ('tests', 'failures', 'errors', 'skipped')}
    result['passed'] = result['tests'] - result['failures'] - result['errors'] - result['skipped']
    result['skipped_cases'] = [{'name': c.get('classname') + '::' + c.get('name'), 'reason': c.find('skipped').get('message')} for c in suite.findall('testcase') if c.find('skipped') is not None]
    return result
